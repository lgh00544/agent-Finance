## ② 超时修复落地核验（15:45）——实现达标，但我查出 2 个待评估副作用

### ✅ 实现完全符合要求（未提交，`akshare_source.py` 41+/4-）
```python
def _accepts_timeout(func) -> bool:      # :341-352 用 inspect.signature 预判，含 VAR_KEYWORD 处理；
    ...                                   # 取不到签名按「不接受」处理 → 替代旧的 except TypeError 兜底
def _call_with_timeout(self, func, *args, **kwargs):   # :1191-1214
    if _accepts_timeout(func):
        return func(*args, **kwargs, timeout=settings.datasource_timeout)
    previous = socket.getdefaulttimeout()
    socket.setdefaulttimeout(settings.datasource_timeout)      # 兜底 1：新建连接超时
    pool = ThreadPoolExecutor(max_workers=1)
    try:
        future = pool.submit(func, *args, **kwargs)
        return future.result(timeout=settings.datasource_timeout + 1)   # 兜底 2：线程等待上限
    except FuturesTimeout:
        raise DataSourceError("数据源 %s 超时 %ss 无返回" % (...)) from None
    finally:
        socket.setdefaulttimeout(previous)                      # 恢复全局超时 ✓
```
